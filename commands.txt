##compiling

#preprocess
gcc-4.9 -Wall -Wextra -Werror -Wmissing-prototypes -pedantic -x c -std=c11 -CC -nostdlib -nodefaultlibs -U '__GNUC__' -include '/workspace/dist/includes/kcc_settings.h' -I 'libc/src/' -E -iquote '.' -iquote 'libc/src' -I '/workspace/dist/includes' 'libc/src/ctype.c' -o 'tmp-kcc-XFadsQa'
#parse
/workspace/dist/cparser tmp-kcc-XFadsQa --trueName 'libc/src/ctype.c' | /workspace/dist/xml-to-k > tmp-kcc-kOVXZs5
#command
krun --output pretty --output-file tmp-kcc-r0Z4LWY -d /workspace/dist/c11-translation-kompiled -cOPTIONS="(SetItem(NOLINK) .Set)" -cOBJ1=".Bag" -cOBJ2=".Bag" --parser cat tmp-kcc-kOVXZs5

#command with exit code 
krun --output pretty --output-file tmp-kcc-r0Z4LWZ -d /workspace/dist/c11-translation-kompiled --exit-code "<result-value> _:Float </result-value>" -cOPTIONS="(SetItem(NOLINK) .Set)" -cOBJ1=".Bag" -cOBJ2=".Bag" --parser cat tmp-kcc-kOVXZs5
